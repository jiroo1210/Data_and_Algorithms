import re

def check_number(number):
    if not re.match("^0[0-9]{8}$", number):
        return False

    weight = [3, 7, 1, 3, 7, 1, 3, 7]
    result = 0
    for pos in range(8):
        result += weight[pos] * int(number[pos])

    last = (10 - result % 10) % 10
    return int(number[-1]) == last
